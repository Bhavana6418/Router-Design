module router_sync_tb();
reg detect_add,write_enb_reg,clock,resetn,read_enb_0,read_enb_1,read_enb_2,empty_0,empty_1,empty_2,full_0,full_1,full_2;
reg [1:0]data_in;
wire vld_out_0,vld_out_1,vld_out_2;
wire [2:0]write_enb;
wire fifo_full,soft_reset_0,soft_reset_1,soft_reset_2;

router_sync DUT(detect_add,write_enb_reg,clock,resetn,read_enb_0,read_enb_1,read_enb_2,empty_0,empty_1,empty_2,full_0,full_1,full_2,
data_in,vld_out_0,vld_out_1,vld_out_2,write_enb,fifo_full,soft_reset_0,soft_reset_1,soft_reset_2);

initial
begin
clock=1'b0;
forever #5 clock=~clock;
end

task rst();
begin
  @(negedge clock)
     resetn=1'b0;
@(negedge clock)
     resetn=1'b1;
end
endtask

initial
begin
{read_enb_0,read_enb_1,read_enb_2,empty_0,empty_1,empty_2,full_0,full_1,full_2}=0;
rst();
@(negedge clock)
   detect_add=1'b1;
   data_in=2'b10;
@(negedge clock)
   detect_add=1'b0;
   write_enb_reg=1'b1;
@(negedge clock)
{full_0,full_1,full_2}=3'b001;
@(negedge clock)
{empty_0,empty_1,empty_2}=3'b110;
#500;
{read_enb_0,read_enb_1,read_enb_2}=3'b000;
#1000 $finish;
end
endmodule
