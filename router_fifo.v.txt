module router_fifo(clock,resetn,write_enb,soft_reset,read_enb,data_in,lfd_state,data_out,full,empty);
input clock,resetn,write_enb,soft_reset,read_enb;
input [7:0]data_in;
input lfd_state;
output reg [7:0]data_out;
output full,empty;
reg [4:0]wptr,rptr;
reg [8:0]mem[15:0];
reg[6:0]count;
integer i,j;
reg temp;

always@(posedge clock)
begin
	if(!resetn)
		temp<=0;
	else
		temp<=lfd_state;
		end

//write logic//
always@(posedge clock)
begin
	if(!resetn)
	begin
		wptr<=0;
		for(i=0;i<16;i=i+1)
		begin
			mem[i]<=0;
		end
	end

       else if(soft_reset)
          begin
            for(j=0;j<16;j=j+1)
	      
                 mem[j]<=0;
              
          end
       else if((write_enb)&&(!full))
          begin
	      {mem[wptr[3:0]][8],mem[wptr[3:0]][7:0]}<={temp,data_in};
	      wptr<=wptr+1;
           end
end

//read logic//
always@(posedge clock)
   begin
	   if(!resetn)
		    begin
			    rptr<=0;
			    data_out<=8'h0;
	            end
           else if(soft_reset)
                  data_out<=8'hz;
           /*else if((count==0) && (data_out!=0))
	          data_out<=8'hz;*/
           else if((read_enb) && (!empty))
			  begin
		   data_out<=mem[rptr[3:0]];
			rptr<=rptr+1;
			end
   end

//counter logic//
/* always@(posedge clock)
   begin
     if(!resetn)
        count<=0;
     else if(soft_reset)
	     begin
           count<=0;
        end
	  
	    else if(mem[rptr[3:0]][8]==1'b1)
		      count<=mem[rptr[3:0]][7:2]+1'b1;
		  else if((read_enb) && (!empty))
              count<=count-1;
				  else 
				  count<=count;
	   
      
    end
	 */
	 
	 always@(posedge clock)
   begin
     if(!resetn)
        count<=0;
     else if(soft_reset)
	                count<=0;
        
	  
	    else if((read_enb) && (!empty))
          begin
		 if((mem[rptr[3:0]][8]==1'b1))
		      count<=mem[rptr[3:0]][7:2]+5'b1;
		  else if(count!=0)
              count<=count-7'b1;
				  else 
				  count<=count;
	   end
      
    end

assign full=(wptr==5'd16)&&(rptr==0);
assign empty=(wptr==rptr);
endmodule
