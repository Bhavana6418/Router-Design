module router_fifo_tb();
reg clock,resetn,write_enb,soft_reset,read_enb;
reg [7:0]data_in;
reg lfd_state;
wire [7:0]data_out;
wire full,empty;
integer k;
router_fifo DUT(clock,resetn,write_enb,soft_reset,read_enb,data_in,lfd_state,data_out,full,empty);

initial
begin
  clock=1'b0;
forever #5 clock=~clock;
end

task reset();
begin
@(negedge clock)
resetn=1'b0;
@(negedge clock)
resetn=1'b1;
end
endtask
 
task soft();
begin
@(negedge clock)
soft_reset=1'b1;
@(negedge clock)
soft_reset=1'b0;
end
endtask

task write();
    reg[7:0]payload_data,parity,header;
	 reg[5:0]payload_len;
	 reg[1:0]addr;
	 begin
	 @(negedge clock)
	 parity=0;
	 payload_len=6'd12;
	 addr=2'b01;
	 header={payload_len,addr};
	 data_in=header;
	 lfd_state=1'b1;
	 write_enb=1'b1;
	 parity=parity^header;
	 for(k=0;k<payload_len;k=k+1)
	 begin
	 @(negedge clock)
	 lfd_state=1'b0;
	 payload_data={$random}%256;
	 data_in=payload_data;
	 parity=parity^payload_data;
	 
	 end
	 @(negedge clock)
	 data_in=parity;
	  @(negedge clock)
	  write_enb=1'b0;
	   
	  
	 end
	 endtask
	 
	 task read();
	  begin
	  @(negedge clock)
	    read_enb=1'b1;
		 end
	 endtask

   initial
	begin
	reset();
	soft();
	write();
	read();
	#1000 $finish;
	end
endmodule
