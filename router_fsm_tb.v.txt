module router_fsm_tb();
  reg clock,resetn,pkt_valid,parity_done,soft_reset_0,soft_reset_1,soft_reset_2,fifo_full,low_pkt_valid,fifo_empty_0,fifo_empty_1,fifo_empty_2;
  reg [1:0]data_in;
  wire detect_add,ld_state,laf_state,full_state,write_enb_reg,rst_int_reg,lfd_state,busy;
  router_fsm DUT(clock,resetn,pkt_valid,parity_done,soft_reset_0,soft_reset_1,soft_reset_2,fifo_full,low_pkt_valid,fifo_empty_0,fifo_empty_1,fifo_empty_2,data_in,
  detect_add,ld_state,laf_state,full_state,write_enb_reg,rst_int_reg,lfd_state,busy);
  
  initial
  begin
  clock=1'b0;
  forever #5 clock=~clock;
  end
  
  task rst();
  begin
  @(negedge clock)
      resetn=1'b0;
	@(negedge clock)
      resetn=1'b1;
  end
endtask

begin

   fifo_empty_1=1'b1;
@(negedge clock)

   fifo_full=1'b0;
    pkt_valid=1'b0;
@(negedge clock)
@(negedge clock)
  fifo_full=1'b0;
end
endtask

begin

  fifo_full=1'b1;
@(negedge clock)
   fifo_full=1'b0;
@(negedge clock)
   parity_done=1'b0;
   low_pkt_valid=1'b1;
@(negedge clock)

end
endtask

begin

  fifo_full=1'b1;

task t4();
begin
@(negedge clock)
   pkt_valid=1'b1;
   data_in=2'b01;
   fifo_empty_1=1'b1;
@(negedge clock)

   fifo_full=1'b0;

  fifo_full=1'b1;

 fifo_full=1'b0;
