module router_fsm(input clock,resetn,pkt_valid,parity_done,soft_reset_0,soft_reset_1,soft_reset_2,fifo_full,low_pkt_valid,fifo_empty_0,fifo_empty_1,fifo_empty_2,input [1:0]data_in,output detect_add,ld_state,laf_state,full_state,write_enb_reg,rst_int_reg,lfd_state,busy);
      parameter da=3'b000,
	   lfd=3'b001,
		wte=3'b010,
		ld=3'b011,
		lp=3'b100,
		ffs=3'b101,
		laf=3'b110,
		cpe=3'b111;
	reg [2:0]state,nextstate;
	always@(posedge clock)
	begin
		if(!resetn)
			state<=da;
		else
			state<=nextstate;
         end
	 always@(*)
	 begin
		 nextstate=da;
		 case({state})
	
			 da:if((pkt_valid)&&(data_in[1:0]==0)&&(fifo_empty_0)||(pkt_valid)&&(data_in[1:0]==1)&&(fifo_empty_1)||(pkt_valid)&&(data_in[1:0]==2)&&(fifo_empty_2))
              nextstate=lfd;
				 else if((pkt_valid)&(data_in[1:0]==0)&&(!fifo_empty_0)||(pkt_valid)&&(data_in[1:0]==1)&&(!fifo_empty_1)||(pkt_valid)&&(data_in[1:0]==2)&&(!fifo_empty_2))
				   nextstate=wte;
				 else
					nextstate=da;
			 wte:if((fifo_empty_0)&&(data_in[1:0]==0)||(fifo_empty_1)&&(data_in[1:0]==1)||(fifo_empty_2)&&(data_in[1:0]==2))
        		nextstate=lfd;
              else
                  nextstate=wte;
          lfd:nextstate=ld;
           ld:if(fifo_full)
              nextstate=ffs;
              else if((!fifo_full)&&(!pkt_valid))
               nextstate=lp;
             else
             nextstate=ld;
			lp:
                 nextstate=cpe;
          ffs:if(!fifo_full)
              nextstate=laf;
             else
              nextstate=ffs;
			laf:
			if((!parity_done)&&(!low_pkt_valid))
			   nextstate=ld;
				else if((!parity_done)&&(low_pkt_valid))
				  nextstate=lp;
				else if(parity_done)
             nextstate=da;
             else
				 nextstate=laf;
           cpe:if(!fifo_full)
              nextstate=da;	
               else if(fifo_full)
                 nextstate=ffs;
               else
             nextstate=cpe;
     endcase
  end
assign busy=((state==lfd)||(state==wte)||(state==lp)||(state==cpe)||(state==ffs)||(state==laf));
assign lfd_state=(state==lfd);
assign write_enb_reg=((state==lp)||(state==ld)||(state==laf));
assign rst_int_reg=(state==cpe);
assign ld_state=((state==ld)||(state==lp));
assign full_state=(state==ffs);
assign detect_add=(state==da)?1'b1:1'b0;
assign laf_state=(state==laf)?1'b1:1'b0;

endmodule
